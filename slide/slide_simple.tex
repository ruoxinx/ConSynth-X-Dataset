\begin{frame}{Motivation: Performance Drops Under Extreme Conditions}

\begin{columns}[T]
\begin{column}{0.55\textwidth}
\textbf{Problem:}
\begin{itemize}
    \item Construction CV models trained on clear-weather data
    \item 40--50\% mAP drop under rain, snow, night, fog
    \item Existing datasets lack systematic condition variation
\end{itemize}

\vspace{0.8em}
\textbf{Prior work:}
\begin{itemize}
    \item Ding et al.\ (2024): UIA-YOLOv5 + NST, ExtCon 506 images, +8.21\% mAP
    \item \textbf{Gaps:} small scale, NST only, no snow/night/scale
\end{itemize}

\vspace{0.8em}
\textbf{ConSynth-X fills these gaps:}
\begin{itemize}
    \item 125,000+ images, 3 complementary pipelines
    \item 5 condition categories + multi-metric validation
\end{itemize}
\end{column}
\begin{column}{0.43\textwidth}
\centering
\small
\begin{tabular}{lc}
\toprule
\textbf{Condition} & \textbf{mAP@0.5} \\
\midrule
Original & 0.819 \\
Weather & 0.446 {\color{warnred}($-$46\%)} \\
Night & 0.454 {\color{warnred}($-$45\%)} \\
\midrule
+Augmented & 0.810 {\color{goodgreen}(+82\%)} \\
\bottomrule
\end{tabular}

\vspace{0.5em}
{\scriptsize SODA dataset, YOLOv8, style transfer augmentation}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Three Complementary Augmentation Pipelines}

\begin{columns}[T]
\begin{column}{0.32\textwidth}
\textbf{1. Style Transfer}\\[0.3em]
{\small
VGG19 feature matching\\
+ MiDaS depth estimation\\
+ Physics particle overlay\\
+ SSIM filtering [0.5, 0.95]\\[0.5em]
{\color{goodgreen}Strong weather effect}\\
{\color{warnred}Texture artifacts}\\
Classifier accuracy: 62--95\%
}
\end{column}
\begin{column}{0.32\textwidth}
\textbf{2. IP2P Diffusion}\\[0.3em]
{\small
InstructPix2Pix (SD 1.5)\\
+ ControlNet (Canny edges)\\
+ Physics particle overlay\\
+ SSIM + LPIPS filtering\\[0.5em]
{\color{goodgreen}Texture faithful (H=0.86)}\\
{\color{warnred}Subtle weather effect}\\
Classifier accuracy: 26--64\%
}
\end{column}
\begin{column}{0.32\textwidth}
\textbf{3. Night / Fog / Scale}\\[0.3em]
{\small
CycleGAN-Turbo (night)\\
Koschmieder + DA V2 (fog)\\
FLUX.1-Fill-dev (outpainting)\\[0.5em]
Night: {\color{goodgreen}FID $-$34\%}\\
Night: {\color{warnred}over-smoothed DCT}\\
Scale: 44\% success rate
}
\end{column}
\end{columns}

\vspace{0.8em}
\centering
{\small \textbf{Key insight:} Style transfer and diffusion are \textbf{complementary}, not competing.\\
Use style transfer for robustness testing, diffusion for natural training data.}
\end{frame}

\begin{frame}{Validation Framework: 5 Complementary Approaches}

\centering
\small
\begin{tabular}{p{3.8cm} p{3.5cm} p{3.5cm} p{3cm}}
\toprule
\textbf{Approach} & \textbf{What it measures} & \textbf{Tool} & \textbf{Result} \\
\midrule
1. Realism Detection & Looks like real photo? & UnivFD (CVPR 2023) & $>$99.5\% fooling \\[0.3em]
2. Weather Recognizability & Weather effect visible? & SigLIP2 classifier & 62--95\% (ST) \\[0.3em]
3. Distributional Fidelity & Close to real weather? & FID/KID vs ACDC & Night: FID $-$34\% \\[0.3em]
4. Texture Fidelity & Micro-texture preserved? & Dempster-Shafer fusion & Diff: H=0.86 \\[0.3em]
5. Feature-space Distance & Embedding shift? & Mahalanobis (CLIP) & Night: 38\% gap closed \\
\bottomrule
\end{tabular}

\vspace{0.5em}
{\footnotesize Each approach probes a different quality dimension.\\Cross-validation reveals the realism--recognizability trade-off.}
\end{frame}

\begin{frame}{Cross-Validation Synthesis: The Trade-Off}

\centering
\small
\begin{tabular}{lcccc}
\toprule
\textbf{Metric} & \textbf{Style Transfer} & \textbf{Diffusion} & \textbf{Night} & \textbf{Outpainting} \\
\midrule
Realism (fooling $\uparrow$) & $>$99.5\% & \textbf{100\%} & 100\% & 100\% \\
Weather recog.\ ($\uparrow$) & \textbf{62--95\%} & 26--64\% & N/A & N/A \\
FID vs.\ ACDC ($\downarrow$) & 192--228 & 194--219 & \textbf{179} & --- \\
Texture $H$ ($\uparrow$) & 0.00 & \textbf{0.86} & 0.02 & --- \\
\bottomrule
\end{tabular}

\vspace{1em}
\begin{columns}[T]
\begin{column}{0.48\textwidth}
\textbf{Realism--Recognizability Trade-off:}
\begin{itemize}
    \item Diffusion = realistic texture, subtle weather
    \item Style Transfer = obvious weather, texture artifacts
    \item \textbf{Complementary, not competing}
\end{itemize}
\end{column}
\begin{column}{0.48\textwidth}
\textbf{Recommendation:}
\begin{itemize}
    \item \textbf{Training data:} use diffusion (natural texture)
    \item \textbf{Robustness testing:} use style transfer (extreme weather)
    \item \textbf{Night:} best semantic shift, known texture limitation
\end{itemize}
\end{column}
\end{columns}
\end{frame}
